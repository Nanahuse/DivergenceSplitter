cmake_minimum_required(VERSION 3.27)
project(DivergenceSplitterSizeProbe VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This experiment requires Windows x64 and MSVC.")
endif()
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Use the windows-release Ninja preset.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
include(GNUInstallDirs)
include(CTest)

find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets Network WebSockets)
find_package(OpenCV 4.12 REQUIRED COMPONENTS core imgcodecs imgproc videoio img_hash)
qt_add_executable(size-probe main.cpp)
# A console subsystem keeps smoke-test diagnostics available in CI.
set_target_properties(size-probe PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(size-probe PRIVATE Qt6::Widgets Qt6::Network Qt6::WebSockets ${OpenCV_LIBS})
target_include_directories(size-probe PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_options(size-probe PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(size-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)

set(NDI_SDK_DIR "" CACHE PATH "Official NDI SDK installation; empty means unverified NDI")
set(NDI_LICENSE_FILE "" CACHE FILEPATH "License agreement from the same official SDK")
if(NDI_SDK_DIR)
    find_path(NDI_INCLUDE_DIR Processing.NDI.Lib.h PATHS "${NDI_SDK_DIR}/Include" NO_DEFAULT_PATH REQUIRED)
    find_file(NDI_RUNTIME Processing.NDI.Lib.x64.dll PATHS "${NDI_SDK_DIR}/Bin/x64" NO_DEFAULT_PATH REQUIRED)
    find_file(NDI_NOTICES Processing.NDI.Lib.Licenses.txt PATHS "${NDI_SDK_DIR}/Bin/x64" NO_DEFAULT_PATH REQUIRED)
    if(NOT EXISTS "${NDI_LICENSE_FILE}" OR IS_DIRECTORY "${NDI_LICENSE_FILE}")
        message(FATAL_ERROR "Set NDI_LICENSE_FILE to the agreement shipped with this SDK.")
    endif()
    target_include_directories(size-probe PRIVATE "${NDI_INCLUDE_DIR}")
    target_compile_definitions(size-probe PRIVATE PROBE_WITH_NDI=1)
    add_custom_command(TARGET size-probe POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${NDI_RUNTIME}" "$<TARGET_FILE_DIR:size-probe>")
    install(FILES "${NDI_RUNTIME}" DESTINATION bin)
    install(FILES "${NDI_NOTICES}" "${NDI_LICENSE_FILE}" DESTINATION licenses/ndi)
    set(PROBE_NDI_STATUS "included; initialization must pass smoke test")
    set(PROBE_NDI_TEST_ARG --require-ndi)
else()
    target_compile_definitions(size-probe PRIVATE PROBE_WITH_NDI=0)
    set(PROBE_NDI_STATUS "UNVERIFIED: official SDK/runtime not supplied; NDI bytes excluded")
endif()

if(BUILD_TESTING)
    add_test(NAME dependency-smoke COMMAND size-probe --self-test ${PROBE_NDI_TEST_ARG})
    set_tests_properties(dependency-smoke PROPERTIES
        TIMEOUT 45
        ENVIRONMENT "QT_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins")
endif()

install(TARGETS size-probe RUNTIME DESTINATION bin)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
include(InstallRequiredSystemLibraries)
qt_generate_deploy_app_script(TARGET size-probe OUTPUT_SCRIPT qt_deploy_script NO_COMPILER_RUNTIME)
install(SCRIPT "${qt_deploy_script}")
# Qt's Windows deploy tool handles Qt plugins. Resolve other DLL dependencies
# after deployment, including transitive dependencies of those plugins and NDI.
configure_file(DeployDependencies.cmake.in DeployDependencies.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/DeployDependencies.cmake")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE" DESTINATION .)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/DISTRIBUTION-NOTICE.txt" DESTINATION .)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/MSVC-NOTICE.txt" DESTINATION licenses/msvc)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ndi-status.txt" "${PROBE_NDI_STATUS}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ndi-status.txt" DESTINATION .)
