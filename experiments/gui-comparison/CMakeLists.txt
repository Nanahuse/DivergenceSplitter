cmake_minimum_required(VERSION 3.27)
set(GUI_CANDIDATE "none" CACHE STRING "none, qt, qtquick, wx, imgui, webview2")
set_property(CACHE GUI_CANDIDATE PROPERTY STRINGS none qt qtquick wx imgui webview2)
set(GUI_QT_FEATURES "standard" CACHE STRING "standard or lean Qt manifest feature")
set_property(CACHE GUI_QT_FEATURES PROPERTY STRINGS standard lean)
option(GUI_QT_OPTIMIZE_DEPLOY "Skip unused Qt raster-Widgets deployment components" OFF)
if(NOT GUI_QT_FEATURES MATCHES "^(standard|lean)$")
    message(FATAL_ERROR "Unknown Qt feature configuration")
endif()
if((NOT GUI_QT_FEATURES STREQUAL "standard" OR GUI_QT_OPTIMIZE_DEPLOY) AND NOT GUI_CANDIDATE STREQUAL "qt")
    message(FATAL_ERROR "Qt configuration options require GUI_CANDIDATE=qt")
endif()
if(GUI_QT_OPTIMIZE_DEPLOY AND NOT GUI_QT_FEATURES STREQUAL "lean")
    message(FATAL_ERROR "Optimized deployment is tested with lean Qt only")
endif()
if(NOT GUI_CANDIDATE STREQUAL "none")
    if(NOT GUI_CANDIDATE MATCHES "^(qt|qtquick|wx|imgui|webview2)$")
        message(FATAL_ERROR "Unknown GUI candidate")
    endif()
    set(gui_feature "${GUI_CANDIDATE}")
    if(GUI_CANDIDATE STREQUAL "qt" AND GUI_QT_FEATURES STREQUAL "lean")
        set(gui_feature qt-lean)
    endif()
    set(VCPKG_MANIFEST_FEATURES "${gui_feature}" CACHE STRING "Candidate feature" FORCE)
endif()
project(DivergenceSplitterGuiComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
include(CTest)
find_package(Threads REQUIRED)
add_library(comparison-mock common/mock.cpp)
target_include_directories(comparison-mock PUBLIC common)
target_link_libraries(comparison-mock PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(comparison-mock PRIVATE /W4 /permissive- /utf-8)
endif()
if(BUILD_TESTING)
    add_executable(comparison-mock-tests tests/mock_tests.cpp)
    target_link_libraries(comparison-mock-tests PRIVATE comparison-mock)
    add_test(NAME common-mock-contract COMMAND comparison-mock-tests)
    set_tests_properties(common-mock-contract PROPERTIES TIMEOUT 15)
endif()
if(NOT GUI_CANDIDATE STREQUAL "none")
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
        message(FATAL_ERROR "GUI comparison requires Windows x64 / MSVC / Release")
    endif()
    find_package(OpenCV 4.12 REQUIRED COMPONENTS core imgproc imgcodecs img_hash videoio)
    add_library(comparison-support common/images.cpp support/app.cpp)
    target_include_directories(comparison-support PUBLIC support ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(comparison-support PUBLIC comparison-mock ${OpenCV_LIBS} psapi)
    target_compile_definitions(comparison-support PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_compile_options(comparison-support PRIVATE /W4 /utf-8)
    set(gui_target comparison-${GUI_CANDIDATE})
    if(GUI_CANDIDATE STREQUAL "qtquick")
        find_package(Qt6 6.11 REQUIRED COMPONENTS Quick QuickControls2 QmlTools)
        # Qt's target finalizer records QML imports for the deployment script.
        qt_add_executable(${gui_target} qtquick/main.cpp)
    else()
        add_executable(${gui_target} ${GUI_CANDIDATE}/main.cpp)
    endif()
    target_link_libraries(${gui_target} PRIVATE comparison-support)
    target_compile_options(${gui_target} PRIVATE /W4 /permissive- /utf-8)
    if(GUI_CANDIDATE STREQUAL "qtquick")
        # Let Qt recognize the existing DPI/long-path manifest instead of creating another identity.
        target_sources(${gui_target} PRIVATE support/app.manifest)
    else()
        target_link_options(${gui_target} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/support/app.manifest")
    endif()
    if(GUI_CANDIDATE STREQUAL "qtquick")
        set_target_properties(${gui_target} PROPERTIES AUTOMOC ON)
        target_link_libraries(${gui_target} PRIVATE Qt6::Quick Qt6::QuickControls2)
        qt_policy(SET QTP0001 NEW)
        set_source_files_properties(qtquick/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
        qt_add_qml_module(${gui_target} URI Comparison VERSION 1.0
            QML_FILES qtquick/Main.qml)
        if(BUILD_TESTING)
            add_test(NAME qtquick-qml-static-contract COMMAND Qt6::qmllint --bare --max-warnings 0
                -I "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/qml"
                "${CMAKE_CURRENT_SOURCE_DIR}/qtquick/Main.qml")
            set_tests_properties(qtquick-qml-static-contract PROPERTIES TIMEOUT 30)
        endif()
    elseif(GUI_CANDIDATE STREQUAL "qt")
        find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets)
        target_link_libraries(${gui_target} PRIVATE Qt6::Widgets)
        if(BUILD_TESTING)
            find_package(Qt6 6.11 REQUIRED COMPONENTS Network)
            add_executable(qt-runtime-check qt/runtime_check.cpp)
            target_link_libraries(qt-runtime-check PRIVATE Qt6::Widgets Qt6::Network)
            target_compile_options(qt-runtime-check PRIVATE /W4 /utf-8)
            # This test executable is kept outside the measured distribution.
        endif()
    elseif(GUI_CANDIDATE STREQUAL "webview2")
        find_package(unofficial-webview2 CONFIG REQUIRED)
        find_package(nlohmann_json CONFIG REQUIRED)
        target_link_libraries(${gui_target} PRIVATE unofficial::webview2::webview2 nlohmann_json::nlohmann_json
            ole32 comdlg32 shlwapi shell32 crypt32)
        target_sources(${gui_target} PRIVATE webview2/protocol.cpp)
        find_program(NPM_EXECUTABLE npm.cmd REQUIRED)
        set(web_ui "${CMAKE_CURRENT_SOURCE_DIR}/webview2/ui")
        file(GLOB_RECURSE web_sources CONFIGURE_DEPENDS "${web_ui}/src/*")
        add_custom_command(OUTPUT "${web_ui}/dist/index.html"
            COMMAND ${NPM_EXECUTABLE} ci --no-audit --no-fund
            COMMAND ${NPM_EXECUTABLE} run build
            DEPENDS ${web_sources} "${web_ui}/package.json" "${web_ui}/package-lock.json"
                "${web_ui}/index.html" "${web_ui}/tsconfig.json" "${web_ui}/vite.config.ts"
            WORKING_DIRECTORY "${web_ui}" VERBATIM)
        add_custom_target(comparison-web-assets DEPENDS "${web_ui}/dist/index.html")
        add_dependencies(${gui_target} comparison-web-assets)
        add_custom_command(TARGET ${gui_target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${web_ui}/dist" "$<TARGET_FILE_DIR:${gui_target}>/ui")
        install(DIRECTORY "${web_ui}/dist/" DESTINATION bin/ui)
        if(BUILD_TESTING)
            add_executable(webview2-protocol-tests webview2/protocol_tests.cpp webview2/protocol.cpp)
            target_link_libraries(webview2-protocol-tests PRIVATE comparison-mock nlohmann_json::nlohmann_json)
            target_compile_options(webview2-protocol-tests PRIVATE /W4 /permissive- /utf-8)
            add_test(NAME webview2-protocol-contract COMMAND webview2-protocol-tests)
            add_test(NAME webview2-typescript-contract COMMAND ${NPM_EXECUTABLE} test -- --reporter=verbose)
            set_tests_properties(webview2-typescript-contract PROPERTIES WORKING_DIRECTORY "${web_ui}" TIMEOUT 30)
        endif()
    elseif(GUI_CANDIDATE STREQUAL "wx")
        find_package(wxWidgets CONFIG REQUIRED)
        target_link_libraries(${gui_target} PRIVATE wx::core wx::base wx::adv)
    else()
        find_package(imgui CONFIG REQUIRED)
        target_link_libraries(${gui_target} PRIVATE imgui::imgui d3d11 dxgi dwmapi comdlg32 ole32)
    endif()
    foreach(dataset small large)
        set(extra_args "")
        if(dataset STREQUAL "large")
            set(extra_args --large)
        endif()
        add_test(NAME ${GUI_CANDIDATE}-${dataset}-gui-contract COMMAND ${gui_target}
            --self-test ${extra_args} --report "${CMAKE_CURRENT_BINARY_DIR}/${dataset}-gui.json"
            --capture "${CMAKE_CURRENT_BINARY_DIR}/${dataset}-gui.png")
        set_tests_properties(${GUI_CANDIDATE}-${dataset}-gui-contract PROPERTIES TIMEOUT 60)
        if(GUI_CANDIDATE MATCHES "^(qt|qtquick)$")
            set_tests_properties(${GUI_CANDIDATE}-${dataset}-gui-contract PROPERTIES
                ENVIRONMENT "QT_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins")
        endif()
    endforeach()
    foreach(view diagnostics profile)
        set(view_args --large --seconds 2 --page ${view})
        if(view STREQUAL "profile")
            list(APPEND view_args --dark)
        endif()
        add_test(NAME ${GUI_CANDIDATE}-${view}-capture COMMAND ${gui_target} ${view_args}
            --report "${CMAKE_CURRENT_BINARY_DIR}/${view}-view.json"
            --capture "${CMAKE_CURRENT_BINARY_DIR}/${view}-view.png")
        set_tests_properties(${GUI_CANDIDATE}-${view}-capture PROPERTIES TIMEOUT 20)
        if(GUI_CANDIDATE MATCHES "^(qt|qtquick)$")
            set_tests_properties(${GUI_CANDIDATE}-${view}-capture PROPERTIES
                ENVIRONMENT "QT_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins")
        endif()
    endforeach()
    add_test(NAME ${GUI_CANDIDATE}-exit-during-hold COMMAND ${gui_target} --hold-on-exit --seconds 1
        --report "${CMAKE_CURRENT_BINARY_DIR}/exit-hold.json")
    set_tests_properties(${GUI_CANDIDATE}-exit-during-hold PROPERTIES TIMEOUT 15)
    if(GUI_CANDIDATE MATCHES "^(qt|qtquick)$")
        set_tests_properties(${GUI_CANDIDATE}-exit-during-hold PROPERTIES
            ENVIRONMENT "QT_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins")
    endif()
    if(GUI_CANDIDATE STREQUAL "qtquick")
        set_tests_properties(qtquick-qml-static-contract qtquick-small-gui-contract
            qtquick-large-gui-contract qtquick-diagnostics-capture qtquick-profile-capture
            qtquick-exit-during-hold PROPERTIES
            ENVIRONMENT "QT_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins;QML_IMPORT_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/qml"
            ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin")
    endif()
    install(TARGETS ${gui_target} RUNTIME DESTINATION bin)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
    include(InstallRequiredSystemLibraries)
    if(GUI_CANDIDATE STREQUAL "qtquick")
        qt_generate_deploy_qml_app_script(TARGET ${gui_target} OUTPUT_SCRIPT qt_deploy
            NO_COMPILER_RUNTIME)
        install(SCRIPT "${qt_deploy}")
    elseif(GUI_CANDIDATE STREQUAL "qt")
        set(qt_deploy_options "")
        if(GUI_QT_OPTIMIZE_DEPLOY)
            # The prototype explicitly uses Fusion/raster Widgets, not Qt Quick or TUIO.
            # Keep Qt Network/TLS and every available image-format plugin.
            list(APPEND qt_deploy_options DEPLOY_TOOL_OPTIONS
                --no-system-d3d-compiler --no-system-dxc-compiler
                --network --skip-plugin-types generic,styles)
        endif()
        qt_generate_deploy_app_script(TARGET ${gui_target} OUTPUT_SCRIPT qt_deploy
            NO_COMPILER_RUNTIME ${qt_deploy_options})
        install(SCRIPT "${qt_deploy}")
    endif()
    set(GUI_EXE "${gui_target}.exe")
    configure_file(DeployDependencies.cmake.in DeployDependencies.cmake @ONLY)
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/DeployDependencies.cmake")
    install(FILES ../../LICENSE DESTINATION .)
    install(FILES DISTRIBUTION-NOTICE.txt DESTINATION .)
    install(FILES ../cpp-size-probe/MSVC-NOTICE.txt DESTINATION licenses/msvc)
endif()
